Question 1: Create a base class called SocialMediaUser with properties username and followers, and a method displayProfile() that prints the username and follower count.
Answer: 
#include <iostream>
using namespace std;

class SocialMediaUser
{
public:
    string username;
    int followers;

    SocialMediaUser(string user, int count)
    {
        username = user;
        followers = count;
    }

    void displayProfile()
    {
        cout << "Username: " << username << endl;
        cout << "Followers: " << followers << endl;
    }
};

int main()
{
    SocialMediaUser user1("dhruvil", 1200);

    user1.displayProfile();

    return 0;
}

Question 2: 
Build a YouTuber class that inherits from SocialMediaUser and adds a property channelName and a method uploadVideo(title) that prints 'Video [title] uploaded to [channelName]'.
Answer:
#include <iostream>
using namespace std;

class SocialMediaUser
{
public:
    string username;
    int followers;

    SocialMediaUser(string user, int count)
    {
        username = user;
        followers = count;
    }

    void displayProfile()

        cout << "Followers: " << followers << endl;

{
public:
    string channelName;

    y1.uploadVideo("My First C++ Video");

Question 5: Implement hierarchical inheritance by creating an InstagramInfluencer class that inherits from SocialMediaUser and adds a method postStory(storyTitle) which prints '[username] posted a new story: [storyTitle]'.<br><br><em><strong>Hint:</strong> Think about how SocialMediaUser is the parent for YouTuber, Podcaster, and InstagramInfluencer.</em>
Answer:
#include <iostream>
using namespace std;

class SocialMediaUser
{
public:
    string username;
    int followers;

    SocialMediaUser(string user, int count)
    {
        username = user;
        followers = count;
    }

    void displayProfile()
    {
        cout << "Username: " << username << endl;
        cout << "Followers: " << followers << endl;
    }
};

class YouTuber : public SocialMediaUser
{
public:
    YouTuber(string user, int count)
        : SocialMediaUser(user, count)
    {
    }
};

class Podcaster : public SocialMediaUser
{
public:
    Podcaster(string user, int count)
        : SocialMediaUser(user, count)
    {
    }
};

class InstagramInfluencer : public SocialMediaUser
{
public:
    InstagramInfluencer(string user, int count)
        : SocialMediaUser(user, count)
    {
    }

    void postStory(string storyTitle)
    {
        cout << username << " posted a new story: "
             << storyTitle << endl;
    }
};

int main()
{
    InstagramInfluencer user1("Dhruvil", 3000);

    user1.displayProfile();
    user1.postStory("College Event");

    return 0;
}
