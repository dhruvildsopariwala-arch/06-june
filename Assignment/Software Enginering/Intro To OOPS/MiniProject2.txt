Mini Project 2: “Creator Dashboard Lite”Objective: Create a console app for digital creators to manage content ideas.Trainer Guidelines:Implement Content class (title, platform, views, status).Allow user to add, edit, delete content items.Store data using file handling.

Question 1: Create a Content class with properties: title, platform, views, and status. Write a method to display all details of a Content object.
Answer:
#include <iostream>
using namespace std;

class Content
{
public:
    string title;
    string platform;
    int views;
    string status;

    void display()
    {
        cout << "Title: " << title << endl;
        cout << "Platform: " << platform << endl;
        cout << "Views: " << views << endl;
        cout << "Status: " << status << endl;
    }
};

int main()
{
    Content c;

    c.title = "Travel Vlog";
    c.platform = "YouTube";
    c.views = 1200;
    c.status = "Published";

    c.display();

    return 0;
}

Question 2: Build a console menu that lets users add new content ideas (title, platform, views, status) and save each entry to a text file named content_list.txt.<br><br><em><strong>Hint:</strong> Use file handling to append each new content item to the file.</em>
Answer:
#include <iostream>
#include <fstream>
using namespace std;

class Content
{
public:
    string title;
    string platform;
    int views;
    string status;

    void addContent()
    {
        cout << "Enter title: ";
        getline(cin, title);

        cout << "Enter platform: ";
        getline(cin, platform);

        cout << "Enter views: ";
        cin >> views;
        cin.ignore();

        cout << "Enter status: ";
        getline(cin, status);
    }

    void saveContent()
    {
        ofstream file("content_list.txt", ios::app);

        file << title << endl;
        file << platform << endl;
        file << views << endl;
        file << status << endl;
        file << "------------------" << endl;

        file.close();
    }
};

int main()
{
    Content c;
    int choice;

    do
    {
        cout << "\n1. Add Content" << endl;
        cout << "2. Exit" << endl;
        cout << "Enter choice: ";
        cin >> choice;
        cin.ignore();

        if (choice == 1)
        {
            c.addContent();
            c.saveContent();

            cout << "Content saved successfully!" << endl;
        }
        else if (choice == 2)
        {
            cout << "Program closed." << endl;
        }
        else
        {
            cout << "Invalid choice!" << endl;
        }

    } while (choice != 2);

    return 0;
}

Question 3:
Implement a function to read all content items from content_list.txt and display them in a numbered list, showing title and platform for each.
Answer:
#include <iostream>
#include <fstream>
using namespace std;

int main()
{
    ifstream file("content_list.txt");

    string title, platform, views, status, line;
    int count = 1;

    while (getline(file, title))
    {
        if (title == "Hello")
            continue;

        getline(file, platform);
        getline(file, views);
        getline(file, status);
        getline(file, line);

        cout << count << ". " << title << " - " << platform << endl;

        count++;
    }

    file.close();

    return 0;
}

Question 4: 
Add an option in your console app to let users update the status of any content idea by selecting its number from the displayed list and saving the change back to the file.<br><br><em><strong>Hint:</strong> You will need to read all items, modify the selected one, and overwrite the file.</em>
Answer:
#include <iostream>
#include <fstream>
using namespace std;

class Content
{
public:
    string title;
    string platform;
    int views;
    string status;
};

int main()
{
    Content items[100];
    int count = 0;

    ifstream file("content_list.txt");

    string line;

    while (getline(file, items[count].title))
    {
        if (items[count].title == "Hello")
            continue;

        getline(file, items[count].platform);

        file >> items[count].views;
        file.ignore();

        getline(file, items[count].status);
        getline(file, line);

        count++;
    }

    file.close();

    if (count == 0)
    {
        cout << "No content found." << endl;
        return 0;
    }

    cout << "\nContent List:\n";

    for (int i = 0; i < count; i++)
    {
        cout << i + 1 << ". "
             << items[i].title << " - "
             << items[i].platform
             << " - " << items[i].status << endl;
    }

    int choice;
    cout << "\nEnter content number to update: ";
    cin >> choice;
    cin.ignore();

    if (choice < 1 || choice > count)
    {
        cout << "Invalid number." << endl;
        return 0;
    }

    cout << "Enter new status: ";
    getline(cin, items[choice - 1].status);

    ofstream outFile("content_list.txt");

    for (int i = 0; i < count; i++)
    {
        outFile << items[i].title << endl;
        outFile << items[i].platform << endl;
        outFile << items[i].views << endl;
        outFile << items[i].status << endl;
        outFile << "Hello" << endl;
    }

    outFile.close();

    cout << "Status updated successfully!" << endl;

    return 0;
}

Question 5: 
Add a feature to delete a content item by its number from the list, update the file accordingly, and display the updated list to confirm deletion.
Answer:
#include <iostream>
#include <fstream>
using namespace std;

class Content
{
public:
    string title;
    string platform;
    int views;
    string status;
};

void showContent(Content items[], int count)
{
    cout << "\nContent List:\n";

    for (int i = 0; i < count; i++)
    {
        cout << i + 1 << ". "
             << items[i].title << " - "
             << items[i].platform << " - "
             << items[i].status << endl;
    }
}

int main()
{
    Content items[100];
    int count = 0;

    ifstream file("content_list.txt");
    string line;

    while (getline(file, items[count].title))
    {
        if (items[count].title == "Hello")
            continue;

        getline(file, items[count].platform);

        file >> items[count].views;
        file.ignore();

        getline(file, items[count].status);
        getline(file, line);

        count++;
    }

    file.close();

    if (count == 0)
    {
        cout << "No content found." << endl;
        return 0;
    }

    showContent(items, count);

    int choice;
    cout << "\nEnter content number to delete: ";
    cin >> choice;

    if (choice < 1 || choice > count)
    {
        cout << "Invalid number." << endl;
        return 0;
    }

    for (int i = choice - 1; i < count - 1; i++)
    {
        items[i] = items[i + 1];
    }

    count--;

    ofstream outFile("content_list.txt");

    for (int i = 0; i < count; i++)
    {
        outFile << items[i].title << endl;
        outFile << items[i].platform << endl;
        outFile << items[i].views << endl;
        outFile << items[i].status << endl;
        outFile << "Hello" << endl;
    }

    outFile.close();

    cout << "\nContent deleted successfully!";

    showContent(items, count);

    return 0;
}
