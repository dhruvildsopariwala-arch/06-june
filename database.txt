----------------------------JOin-------------------------
CREATE TABLE db_city(

id int PRIMARY KEY AUTO_INCREMENT,
city_name varchar(50)
)

CREATE TABLE db_emp(

id int PRIMARY KEY AUTO_INCREMENT,
name varchar(50),
city_id int ,
FOREIGN KEY(city_id) REFERENCES db_city(id)
)


1>SELECT  db_emp.id,db_emp.name,db_city.city_name FROM db_emp INNER JOIN db_city ON db_emp.city_id = db_city.id;
2>SELECT  db_emp.id,db_emp.name,db_city.city_name FROM db_emp LEFT JOIN db_city ON db_emp.city_id = db_city.id;
3>SELECT  db_emp.id,db_emp.name,db_city.city_name FROM db_emp RIGHT JOIN db_city ON db_emp.city_id = db_city.id;
